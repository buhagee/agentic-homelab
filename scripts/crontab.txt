# =============================================================================
# crontab.txt — Qacey Platform scheduled jobs (source of truth)
# =============================================================================
# ALL automated jobs are documented here, regardless of scheduler.
# Install cron jobs with: bash scripts/install-crontab.sh
#
# TIMEZONE: Server runs UTC. Australia/Sydney = UTC+10 (AEST) / UTC+11 (AEDT)
#
# ┌──────────────────────────────────────────────────────────────────────────────┐
# │  TIME (UTC)   │  LOCAL (AEST)  │  JOB                      │  SCHEDULER    │
# ├──────────────────────────────────────────────────────────────────────────────┤
# │  */15 * * * * │  every 15 min  │  Memory monitor           │  cron         │
# │  */15 * * * * │  every 15 min  │  Host healthcheck         │  systemd      │
# │  0 14 * * *   │  12:00 AM      │  SkillClaw evolve (LLM)   │  cron         │
# │  0 15 * * *   │  1:00 AM       │  Immich face-to-album     │  cron         │
# │  0 16 * * *   │  2:00 AM       │  DB dump + rsync + B2     │  systemd      │
# └──────────────────────────────────────────────────────────────────────────────┘
#
# =============================================================================
# SYSTEM CRON JOBS (managed by this file — install with: bash scripts/install-crontab.sh)
# =============================================================================

# ── Maintenance ───────────────────────────────────────────────────────────────

# Memory monitor — logs container memory usage every 15 minutes
*/15 * * * * $HOME/projects/qacey-platform/scripts/maintenance/monitor-memory.sh >> /var/log/qacey-cron/monitor-memory.log 2>&1

# ── Immich ────────────────────────────────────────────────────────────────────

# Face-to-album sync — daily at 1:00 AM AEST (15:00 UTC), 1 hour before backup
0 15 * * * $HOME/projects/qacey-platform/scripts/tools/immich-face-to-album/immich-face-to-album.sh >> /var/log/qacey-cron/face-to-album.log 2>&1

# ── SkillClaw ─────────────────────────────────────────────────────────────────

# SkillClaw skill evolution — daily at 12:00 AM AEST (14:00 UTC)
# Drains captured agent sessions and evolves the shared skill library (local 7b
# model, CPU-heavy). Scheduled between the nightly digest (13:00) and
# face-to-album (15:00) so the CPU-bound LLM jobs don't overlap.
# The skillclaw evolve server's own interval is set to 24h (idle); this drives it.
0 14 * * * $HOME/projects/qacey-platform/scripts/maintenance/skillclaw-evolve.sh >> /var/log/qacey-cron/skillclaw-evolve.log 2>&1

# =============================================================================
# SYSTEMD JOBS (managed separately — documented here for visibility)
# =============================================================================
#
#   qacey-backup.timer — nightly at 02:00 AEST (16:00 UTC)
#     1. qacey-db-backup.service  → dumps Immich/Paperless/Mnemo DBs

#     Install:  sudo bash scripts/maintenance/install-backup.sh
#     Status:   systemctl list-timers qacey-backup.timer
